\begin{frame}{A conductance flyvis model}
\srcpath{src/flyvis\_conductance\_optical\_flow/dynamics.py}
\vfill
{\small\textbf{Nominal current flyvis}}
\[ \tau_i \dot v_i = -(v_i - V^{\mathrm{rest}}_i) + \sum_j W_{ij}\,\mathrm{relu}(v_j) + I_i(t) \]
\vfill
{\small\textbf{New conductance flyvis}, $W_{ij}\ge 0$}
\[ \tau_i \dot v_i = -(v_i - V^{\mathrm{rest}}_i) + \sum_j W_{ij}\,\mathrm{relu}(v_j)\,(E_i - v_i) + I_i(t) \]
\vfill
\end{frame}

\begin{frame}{Training the conductance flyvis model}
\srcpath{src/flyvis\_conductance\_optical\_flow/train.py \textbar{} flyvis/results/flow/2000/001/\_meta.yaml}
\vspace{0.45cm}
\scriptsize
\setlength{\itemsep}{1pt}
\begin{itemize}\itemsep1pt
\item \textbf{Stimuli} --- Sintel, \texttt{original\_sampling}; box filter extent 15,
      kernel 13; 19 frames, 3 vertical splits, crop 0.7; flip / rotate $p=0.5$,
      contrast $\sigma\!=\!0.2$, brightness $\sigma\!=\!0.1$, white noise $0.08$
\item \textbf{Rollout} --- 19 frames resampled to $\Delta t = 20$\,ms
      $\Rightarrow$ \textbf{40 forward-Euler steps} per sample, back-propagated through all
\item \textbf{$E_i = V^{\mathrm{rest}}_i + \mathrm{band}$}, excursion $1.0$, bands
      $[1.5, 6.0]$ / $[-6.0, -0.5]$: 66 free variables, \textbf{130 reversals} ($V^{\mathrm{rest}}$ is per type)
\item \textbf{Optimiser} --- Adam, $\mathrm{lr} = 5\times10^{-5}$, batch 4,
      250{,}000 iterations, 65 checkpoints; H100, 5.5\,h
\item \textbf{Loss} $\ell_2$ on decoded flow: $1214.7 \rightarrow 1175.3$,
      \textbf{min 1147.1 at checkpoint 28}
\item \textbf{EPE} $= \big\langle \lVert \hat{\mathbf{f}} - \mathbf{f} \rVert_2 \big\rangle$
      over hexagonal pixels and frames: $5.785 \rightarrow 5.454$,
      \textbf{min 5.390 at checkpoint 28}
\end{itemize}

\begin{tabular}{@{}l@{}}
\toprule
$\mu_{\mathrm{sil}}$ \quad silent anchor \quad \texttt{coeff\_g\_phi\_silent} \\
\midrule
$\mathcal{L}_{\mathrm{sil}} = \mu_{\mathrm{sil}}
 \bigl\lVert\, g_\phi(\tilde v_j, \mathbf{a}_j, v_i, \mathbf{a}_i) \,\bigr\rVert_2 ,
 \qquad \tilde v_j \sim \mathcal{U}[-2, 0]$ \\[6pt]
$\mathrm{relu}(v_j) = 0 \ \Rightarrow\ \mathrm{msg}_{ij} = 0
 \quad\text{for every } E_i, v_i$ \\
\midrule
$\mu_{\mathrm{gain}}$ \quad message gain \quad \texttt{coeff\_f\_theta\_msg\_gain} \\
\midrule
$\mathcal{L}_{\mathrm{gain}} = \mu_{\mathrm{gain}}
 \left\lVert \dfrac{\partial f_\theta/\partial \mathrm{msg}_i
   \;+\; \partial f_\theta/\partial v_i}
  {\bigl\langle (\partial f_\theta/\partial v_i)^2 \bigr\rangle^{1/2}}
 \right\rVert_2$ \\[8pt]
$\tau_i \dot v_i = V^{\mathrm{rest}}_i - v_i + \mathrm{msg}_i + I_i
 \ \Rightarrow\
 \dfrac{\partial f_\theta}{\partial \mathrm{msg}_i} = \dfrac{1}{\tau_i},\quad
 \dfrac{\partial f_\theta}{\partial v_i} = -\dfrac{1}{\tau_i}$ \\[4pt]
$\Rightarrow\ \partial f_\theta/\partial \mathrm{msg}_i
 + \partial f_\theta/\partial v_i = 0 \quad \forall\, \tau_i$ \\[6pt]
$\mathrm{msg}_\theta = \mathrm{msg}/c,\ \
 \partial f_\theta/\partial \mathrm{msg} = c/\tau_i
 \ \Rightarrow\ \dot v_i \text{ unchanged}$ \\
\bottomrule
\end{tabular}
\end{frame}

\begin{frame}{Template readout}

\centering\small
\begin{tabular}{@{}l@{}}
\toprule
affine message ansatz \\
\midrule
$\widehat{\mathrm{msg}}_{ij}(t) = \dfrac{1}{k_i}\,\mathrm{msg}_{ij}(t) + C_{ij}$
 \qquad $k_i$ per postsynaptic neuron \\
\midrule
per-edge template fit, $u_t = \mathrm{relu}(v_j(t))$ \\
\midrule
$\mathrm{msg}_{ij} = W_{ij}\,u_t\,(E_i - v_i) + C_{ij}
 = \underbrace{\tfrac{W_{ij}E_i}{k_i}}_{b_1} u_t
 \;\underbrace{-\tfrac{W_{ij}}{k_i}}_{b_2} (u_t v_i)
 \;+\; \underbrace{C_{ij}}_{b_3}$ \\[6pt]
$\Rightarrow\quad W^{\mathrm{fit}}_{ij} = -b_2, \qquad E_i = -b_1/b_2, \qquad C_{ij} = b_3$ \\
\midrule
per-neuron update fit on $[\,1,\ v_i,\ \mathrm{msg}_i,\ I_i\,]$ \\
\midrule
$\dot v_i = \alpha_0 + \alpha_1 v_i + \alpha_2 \widehat{\mathrm{msg}}_i + \alpha_3 I_i$
 \qquad
$\tau_i = -1/\alpha_1, \quad k_i = \tau_i\alpha_2 = -\alpha_2/\alpha_1$ \\[6pt]
$V^{\mathrm{rest}}_i = \tau_i(\alpha_0 + \alpha_2\beta_i), \qquad
 W^{\mathrm{corrected}}_{ij} = k_i\,W^{\mathrm{fit}}_{ij}$ \\
\bottomrule
\end{tabular}
\end{frame}
