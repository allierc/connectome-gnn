\begin{frame}{A conductance flyvis model}
\srcpath{src/flyvis\_conductance\_optical\_flow/dynamics.py}
\vfill
{\small\textbf{Nominal current flyvis}}
\[ \tau_i \dot v_i = -(v_i - V^{\mathrm{rest}}_i) + \sum_j W_{ij}\,\mathrm{relu}(v_j) + I_i(t) \]
\vfill
{\small\textbf{New conductance flyvis}, $W_{ij}\ge 0$}
\[ \tau_i \dot v_i = -(v_i - V^{\mathrm{rest}}_i) + \sum_j W_{ij}\,\mathrm{relu}(v_j)\,(E_i - v_i) + I_i(t) \]
\vfill
\end{frame}

\begin{frame}{Training the conductance flyvis model}
\srcpath{src/flyvis\_conductance\_optical\_flow/train.py \textbar{} flyvis/results/flow/2000/001/\_meta.yaml}
\scriptsize
\setlength{\itemsep}{1pt}
\begin{itemize}\itemsep1pt
\item \textbf{Stimuli} --- Sintel, \texttt{original\_sampling}; box filter extent 15,
      kernel 13; 19 frames, 3 vertical splits, crop 0.7; flip / rotate $p=0.5$,
      contrast $\sigma\!=\!0.2$, brightness $\sigma\!=\!0.1$, white noise $0.08$
\item \textbf{Rollout} --- 19 frames resampled to $\Delta t = 20$\,ms
      $\Rightarrow$ \textbf{40 forward-Euler steps} per sample, back-propagated through all
\item \textbf{$E_i = V^{\mathrm{rest}}_i + \mathrm{band}$}, excursion $1.0$, bands
      $[1.5, 6.0]$ / $[-6.0, -0.5]$: 66 free variables, \textbf{130 reversals} ($V^{\mathrm{rest}}$ is per type)
\item \textbf{Optimiser} --- Adam, $\mathrm{lr} = 5\times10^{-5}$, batch 4,
      250{,}000 iterations, 65 checkpoints; H100, 5.5\,h
\item \textbf{Loss} $\ell_2$ on decoded flow: $1214.7 \rightarrow 1175.3$,
      \textbf{min 1147.1 at checkpoint 28}
\item \textbf{EPE} $= \big\langle \lVert \hat{\mathbf{f}} - \mathbf{f} \rVert_2 \big\rangle$
      over hexagonal pixels and frames: $5.785 \rightarrow 5.454$,
      \textbf{min 5.390 at checkpoint 28}
\end{itemize}
\vspace{1pt}
